+using Verse;
+
+namespace GiddyUpCavalryCharge
+{
+
+public sealed class CavalryTemplatePairRecord : IExposable
+{
+    public int slotIndex = -1;
+    public string riderLoadId;
+    public string riderLabel;
+    public string mountLoadId;
+    public string mountLabel;
+
+    public CavalryTemplatePairRecord()
+    {
+    }
+
+    public CavalryTemplatePairRecord(int slotIndex, Pawn rider, Pawn mount)
+    {
+        this.slotIndex = slotIndex;
+        CaptureRider(rider);
+        CaptureMount(mount);
+    }
+
+    public void CaptureRider(Pawn pawn)
+    {
+        riderLoadId = pawn != null && !pawn.Destroyed ? pawn.GetUniqueLoadID() : null;
+        riderLabel = pawn != null ? pawn.LabelShortCap : null;
+    }
+
+    public void CaptureMount(Pawn pawn)
+    {
+        mountLoadId = pawn != null && !pawn.Destroyed ? pawn.GetUniqueLoadID() : null;
+        mountLabel = pawn != null ? pawn.LabelShortCap : null;
+    }
+
+    public bool HasRiderSnapshot
+    {
+        get { return !string.IsNullOrEmpty(riderLoadId) || !string.IsNullOrEmpty(riderLabel); }
+    }
+
+    public bool HasMountSnapshot
+    {
+        get { return !string.IsNullOrEmpty(mountLoadId) || !string.IsNullOrEmpty(mountLabel); }
+    }
+
+    public void ExposeData()
+    {
+        Scribe_Values.Look(ref slotIndex, "slotIndex", -1);
+        Scribe_Values.Look(ref riderLoadId, "riderLoadId");
+        Scribe_Values.Look(ref riderLabel, "riderLabel");
+        Scribe_Values.Look(ref mountLoadId, "mountLoadId");
+        Scribe_Values.Look(ref mountLabel, "mountLabel");
+    }
+}
