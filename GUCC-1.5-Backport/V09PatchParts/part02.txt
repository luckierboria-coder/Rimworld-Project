+        if (unit == null)
+        {
+            reason = "GUCC_Error_InvalidPair".Translate();
+            return false;
+        }
+        if (GetActiveFormationChargeForUnit(unit.id) != null)
+        {
+            reason = "GUCC_UnitWindow_EditingLocked".Translate();
+            return false;
+        }
+
+        var clamped = Mathf.Clamp(
+            requested,
+            ChargeConstants.MinimumCavalryMembers,
+            ChargeConstants.MaximumCavalryMembers);
+        if (unit.HighestOccupiedSlotIndex >= clamped)
+        {
+            reason = "GUCC_Error_UnitSizeOccupied".Translate(
+                unit.HighestOccupiedSlotIndex + 1,
+                clamped);
+            return false;
+        }
+
+        if (unit.targetSize != clamped)
+        {
+            unit.targetSize = clamped;
+            FinishRosterEdit(unit);
+        }
+        return true;
+    }

+    public Pawn ResolveTemplatePawn(string loadId)
+    {
+        if (string.IsNullOrEmpty(loadId))
+            return null;
+
+        for (var mapIndex = 0; mapIndex < Find.Maps.Count; mapIndex++)
+        {
+            var pawns = Find.Maps[mapIndex].mapPawns.AllPawnsSpawned;
+            for (var i = 0; i < pawns.Count; i++)
+            {
+                var pawn = pawns[i];
+                if (pawn != null && pawn.GetUniqueLoadID() == loadId)
+                    return pawn;
+            }
+        }
+
+        if (Find.WorldPawns != null)
+        {
+            var pawns = Find.WorldPawns.AllPawnsAliveOrDead;
+            for (var i = 0; i < pawns.Count; i++)
+            {
+                var pawn = pawns[i];
+                if (pawn != null && pawn.GetUniqueLoadID() == loadId)
+                    return pawn;
+            }
+        }
+        return null;
+    }

+    public void GetTemplateAvailability(
+        CavalryUnitTemplateRecord template,
+        out int availableRiders,
+        out int availableMounts,
+        out int availableCompletePairs,
+        out bool commanderAvailable)
+    {
+        availableRiders = 0;
+        availableMounts = 0;
+        availableCompletePairs = 0;
+        commanderAvailable = false;
+        if (template == null)
+            return;
+
+        for (var slotIndex = 0; slotIndex < template.TargetSize; slotIndex++)
+        {
+            var snapshot = template.GetSlot(slotIndex);
+            if (snapshot == null)
+                continue;
+            var rider = ResolveTemplatePawn(snapshot.riderLoadId);
+            var mount = ResolveTemplatePawn(snapshot.mountLoadId);
+            var riderAvailable = IsTemplateRiderAvailable(rider, null);
+            var mountAvailable = IsTemplateMountAvailable(mount, null);
+            if (riderAvailable)
+                availableRiders++;
+            if (mountAvailable)
+                availableMounts++;
+            if (riderAvailable && mountAvailable)
+                availableCompletePairs++;
+            if (slotIndex == 0)
+                commanderAvailable = riderAvailable;
+        }
+    }
