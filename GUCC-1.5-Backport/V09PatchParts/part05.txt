+    private void DrawUnitRow(Rect rect, CavalryUnitRecord unit)
+    {
+        if (unit == null)
+            return;
+
+        var focused = unit.id == focusUnitId;
+        Widgets.DrawBoxSolid(
+            rect,
+            focused
+                ? new Color(0.19f, 0.27f, 0.16f, 0.78f)
+                : new Color(0.12f, 0.12f, 0.12f, 0.72f));
+        Widgets.DrawBox(rect);
+
+        Text.Font = GameFont.Small;
+        Widgets.Label(
+            new Rect(rect.x + 8f, rect.y + 5f, 250f, 24f),
+            unit.DisplayName);
+
+        var commander = unit.CommanderPair != null ? unit.CommanderPair.rider : unit.commander;
+        var commanderText = commander != null && !commander.Destroyed
+            ? commander.LabelShortCap
+            : "GUCC_Manager_MissingCommander".Translate().ToString();
+        Text.Font = GameFont.Tiny;
+        Widgets.Label(
+            new Rect(rect.x + 8f, rect.y + 32f, 350f, 22f),
+            "GUCC_Manager_UnitSummary".Translate(
+                commanderText,
+                unit.CompleteMemberCount,
+                unit.TargetSize,
+                CavalryFormationUtility.Label(unit.formation)));
+        Text.Font = GameFont.Small;
+
+        var buttonWidth = 116f;
+        var gap = 6f;
+        var right = rect.xMax - 8f;
+        var disbandRect = new Rect(right - buttonWidth, rect.y + 18f, buttonWidth, 30f);
+        right -= buttonWidth + gap;
+        var saveRect = new Rect(right - buttonWidth, rect.y + 18f, buttonWidth, 30f);
+        right -= buttonWidth + gap;
+        var manageRect = new Rect(right - buttonWidth, rect.y + 18f, buttonWidth, 30f);
+
+        if (Widgets.ButtonText(manageRect, "GUCC_Manager_Manage".Translate()))
+        {
+            Find.WindowStack.Add(new Window_CavalryUnit(unit));
+            Close();
+            return;
+        }
+
+        if (Widgets.ButtonText(saveRect, "GUCC_Template_Save".Translate()))
+        {
+            var template = ChargeWorldComponent.Current.SaveUnitAsTemplate(unit);
+            if (template != null)
+                Messages.Message(
+                    "GUCC_Template_Saved".Translate(template.DisplayName),
+                    MessageTypeDefOf.TaskCompletion,
+                    false);
+        }
+
+        var locked = ChargeWorldComponent.Current.GetActiveFormationChargeForUnit(unit.id) != null;
+        if (Widgets.ButtonText(disbandRect, "GUCC_UnitWindow_Disband".Translate()) && !locked)
+        {
+            ChargeWorldComponent.Current.DisbandUnit(unit);
+            Messages.Message(
+                "GUCC_Manager_UnitDisbanded".Translate(unit.DisplayName),
+                MessageTypeDefOf.NeutralEvent,
+                false);
+        }
+        if (locked)
+            TooltipHandler.TipRegion(disbandRect, "GUCC_UnitWindow_EditingLocked".Translate());
+    }
